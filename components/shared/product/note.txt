'use client'

import { useState, useRef, useEffect } from 'react'
import Image from 'next/image'
import { Swiper, SwiperSlide } from 'swiper/react'
import { Navigation, Thumbs, FreeMode } from 'swiper/modules'
import type { Swiper as SwiperType } from 'swiper'
import 'swiper/css'
import 'swiper/css/navigation'
import 'swiper/css/thumbs'
import 'swiper/css/free-mode'
import { IIMage } from '@/types'

export default function ProductGallery({ images }: { images: IIMage[] }) {
  const [selectedImage, setSelectedImage] = useState(0)
  const [thumbsSwiper, setThumbsSwiper] = useState<SwiperType | null>(null)
  const [isLightboxOpen, setIsLightboxOpen] = useState(false)
  const mainSwiperRef = useRef<SwiperType | null>(null)

  // Navigation handlers
  const nextImage = () => {
    if (mainSwiperRef.current) {
      mainSwiperRef.current.slideNext()
    }
  }

  const prevImage = () => {
    if (mainSwiperRef.current) {
      mainSwiperRef.current.slidePrev()
    }
  }

  // Keyboard navigation for lightbox
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (!isLightboxOpen) return
      
      if (e.key === 'Escape') {
        setIsLightboxOpen(false)
      } else if (e.key === 'ArrowRight') {
        nextImage()
      } else if (e.key === 'ArrowLeft') {
        prevImage()
      }
    }

    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [isLightboxOpen])

  // Disable body scroll when lightbox is open
  useEffect(() => {
    if (isLightboxOpen) {
      document.body.style.overflow = 'hidden'
    } else {
      document.body.style.overflow = 'auto'
    }
    
    return () => {
      document.body.style.overflow = 'auto'
    }
  }, [isLightboxOpen])

  return (
    <div className="grid grid-cols-1 md:grid-cols-6 gap-4 w-full mx-auto p-4">
      {/* Thumbnail Column - Desktop */}
      <div className="hidden md:flex flex-col order-2 md:order-1 col-span-1">
        <Swiper
          direction="vertical"
          onSwiper={setThumbsSwiper}
          spaceBetween={12}
          slidesPerView={5}
          freeMode={true}
          watchSlidesProgress={true}
          modules={[FreeMode, Thumbs]}
          className="h-[500px] w-full"
        >
          {images.map((image, index) => (
            <SwiperSlide key={index} className="cursor-pointer flex px-2 py-4 justify-center">
              <div 
                className={`
                  relative overflow-hidden rounded-lg transition-all duration-300 border-2 w-full
                  ${selectedImage === index
                    ? 'border-blue-500 scale-105 shadow-md'
                    : 'border-gray-300 opacity-80 hover:opacity-100 hover:border-gray-400'}
                `} 
                style={{ height: '80px' }}
                onClick={() => {
                  setSelectedImage(index)
                  if (mainSwiperRef.current) {
                    mainSwiperRef.current.slideTo(index)
                  }
                }}
              >
                <Image
                  src={image.url}
                  alt={`Thumbnail ${index + 1}`}
                  fill
                  sizes="80px"
                  className="object-cover"
                />
                {/* Selection indicator */}
                {selectedImage === index && (
                  <div className="absolute inset-0  border-blue-400 rounded-lg" />
                )}
              </div>
            </SwiperSlide>
          ))}
        </Swiper>
      </div>

      {/* Thumbnail Row - Mobile */}
      <div className="block md:hidden order-2 mt-4">
        <Swiper
          onSwiper={setThumbsSwiper}
          spaceBetween={10}
          slidesPerView={4}
          freeMode={true}
          watchSlidesProgress={true}
          modules={[FreeMode, Thumbs]}
          className="w-full"
        >
          {images.map((image, index) => (
            <SwiperSlide key={index} className="cursor-pointer flex justify-center">
              <div 
                className={`
                  relative overflow-hidden rounded-lg transition-all duration-300 border-2
                  ${selectedImage === index
                    ? 'border-blue-500 scale-105 shadow-md'
                    : 'border-gray-300 opacity-80 hover:opacity-100 hover:border-gray-400'}
                `} 
                style={{ width: '70px', height: '70px' }}
                onClick={() => {
                  setSelectedImage(index)
                  if (mainSwiperRef.current) {
                    mainSwiperRef.current.slideTo(index)
                  }
                }}
              >
                <Image
                  src={image.url}
                  alt={`Thumbnail ${index + 1}`}
                  fill
                  sizes="70px"
                  className="object-cover"
                />
                {/* Selection indicator */}
                {selectedImage === index && (
                  <div className="absolute inset-0 border-2 border-blue-400 rounded-lg" />
                )}
              </div>
            </SwiperSlide>
          ))}
        </Swiper>
      </div>

      {/* Main Image Container */}
      <div className="relative w-full col-span-5 order-1 md:order-2">
        <div className="relative aspect-square overflow-hidden rounded-xl bg-gray-100 shadow-lg">
          <Swiper
            thumbs={{ swiper: thumbsSwiper && !thumbsSwiper.destroyed ? thumbsSwiper : null }}
            modules={[Navigation, Thumbs]}
            className="main-swiper h-full"
            onSwiper={(swiper) => mainSwiperRef.current = swiper}
            onSlideChange={(swiper) => setSelectedImage(swiper.activeIndex)}
            navigation={{
              nextEl: '.swiper-button-next',
              prevEl: '.swiper-button-prev',
            }}
          >
            {images.map((image, index) => (
              <SwiperSlide key={index}>
                <div 
                  className="relative w-full h-full cursor-zoom-in"
                  onClick={() => setIsLightboxOpen(true)}
                >
                  <Image
                    src={image.url}
                    alt={`Product image ${index + 1}`}
                    fill
                    sizes="(max-width: 768px) 100vw, 60vw"
                    className="object-contain"
                    priority={index === 0}
                  />
                </div>
              </SwiperSlide>
            ))}
          </Swiper>

          {/* Navigation Arrows */}
          {images.length > 1 && (
            <>
              <button
                onClick={prevImage}
                className="swiper-button-prev absolute left-3 top-1/2 -translate-y-1/2 bg-white/90 hover:bg-white rounded-full p-3 shadow-md transition-all duration-200 z-10"
                aria-label="Previous image"
              >
                <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" />
                </svg>
              </button>
              <button
                onClick={nextImage}
                className="swiper-button-next absolute right-3 top-1/2 -translate-y-1/2 bg-white/90 hover:bg-white rounded-full p-3 shadow-md transition-all duration-200 z-10"
                aria-label="Next image"
              >
                <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" />
                </svg>
              </button>
            </>
          )}

          {/* Image Counter */}
          {images.length > 1 && (
            <div className="absolute top-4 right-4 bg-black/70 text-white text-sm px-3 py-1 rounded-full z-10">
              {selectedImage + 1} / {images.length}
            </div>
          )}

          {/* Zoom Indicator */}
          <div className="absolute bottom-4 left-4 bg-black/70 text-white text-sm px-3 py-1 rounded-full z-10 flex items-center gap-1">
            <svg xmlns="http://www.w3.org/2000/svg" className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
            </svg>
            <span>Click to zoom</span>
          </div>
        </div>
      </div>

      {/* Lightbox */}
      {isLightboxOpen && (
        <div className="fixed inset-0 bg-black/90 z-50 flex items-center justify-center p-4">
          <button
            onClick={() => setIsLightboxOpen(false)}
            className="absolute top-4 right-4 text-white bg-black/50 hover:bg-black/70 rounded-full p-2 z-50"
            aria-label="Close lightbox"
          >
            <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>

          <div className="relative w-full max-w-4xl h-full max-h-[80vh]">
         <Swiper
            thumbs={{ swiper: thumbsSwiper && !thumbsSwiper.destroyed ? thumbsSwiper : null }}
            onActiveIndexChange={(swiper) => setSelectedImage(swiper.activeIndex)}
            modules={[Navigation, Thumbs]}
            className="main-swiper h-full"
            onSwiper={(swiper) => mainSwiperRef.current = swiper}
            onSlideChange={(swiper) => setSelectedImage(swiper.activeIndex)}
            navigation={{
              nextEl: '.swiper-button-next',
              prevEl: '.swiper-button-prev',
            }}
          >
            {images.map((image, index) => (
              <SwiperSlide key={index}>
                <div 
                  className="relative w-full h-full cursor-zoom-in"
                  onClick={() => setIsLightboxOpen(true)}
                >
                  <Image
                    src={image.url}
                    alt={`Product image ${index + 1}`}
                    fill
                    sizes="(max-width: 768px) 100vw, 60vw"
                    className="object-contain"
                    priority={index === 0}
                  />
                </div>
              </SwiperSlide>
            ))}
          </Swiper>
            {/* Lightbox Navigation */}
            <button
              onClick={prevImage}
              className="lightbox-button-prev absolute left-4 top-1/2 -translate-y-1/2 text-white bg-black/50 hover:bg-black/70 rounded-full p-3 z-10"
              aria-label="Previous image"
            >
              <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" />
              </svg>
            </button>
            <button
              onClick={nextImage}
              className="lightbox-button-next absolute right-4 top-1/2 -translate-y-1/2 text-white bg-black/50 hover:bg-black/70 rounded-full p-3 z-10"
              aria-label="Next image"
            >
              <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" />
              </svg>
            </button>

            {/* Lightbox Counter */}
            <div className="absolute bottom-4 left-1/2 -translate-x-1/2 bg-black/70 text-white text-sm px-3 py-1 rounded-full">
              {selectedImage + 1} / {images.length}
            </div>
          </div>
        </div>
      )}
    </div>
  )
}
